create table if not exists public.tickets (
  number text primary key,
  tier text not null check (tier in ('Ordinary','VIP','VVIP')),
  code text not null, salt text not null,
  buyer text not null default '',
  status text not null default 'Unused' check (status in ('Unused','Used')),
  created_at bigint not null, used_at bigint,
  printed boolean not null default false
);
alter table public.tickets add column if not exists shared boolean not null default false;
create index if not exists tickets_created_idx on public.tickets (created_at, number);
alter table public.tickets enable row level security;
drop policy if exists "anon full access" on public.tickets;
create policy "anon full access" on public.tickets for all to anon using (true) with check (true);

create table if not exists public.ticket_counters (tier text primary key, last int not null default 0);
alter table public.ticket_counters enable row level security;

-- staff accounts: passwords are stored only as bcrypt hashes in a table anon cannot read
create table if not exists public.users (
  username text primary key,
  role text not null check (role in ('admin','producer','gatekeeper')),
  created_at bigint not null default (extract(epoch from now()) * 1000)::bigint
);
create table if not exists public.user_secrets (
  username text primary key references public.users(username) on delete cascade,
  pw_hash text not null
);
create table if not exists public.sessions (
  token text primary key,
  username text not null references public.users(username) on delete cascade,
  created_at timestamptz not null default now()
);
create table if not exists public.settings (key text primary key, value jsonb not null);
alter table public.users enable row level security;
alter table public.user_secrets enable row level security;
alter table public.sessions enable row level security;
alter table public.settings enable row level security;
drop policy if exists "read staff list" on public.users;
create policy "read staff list" on public.users for select to anon using (true);

create or replace function public._admin(p_token text) returns boolean language sql security definer set search_path = public as $$
  select exists (select 1 from sessions s join users u using (username) where s.token = p_token and u.role = 'admin') $$;
create or replace function public._staff(p_token text) returns boolean language sql security definer set search_path = public as $$
  select exists (select 1 from sessions s join users u using (username) where s.token = p_token and u.role in ('admin','producer')) $$;

create or replace function public.app_login(p_user text, p_pass text) returns json language plpgsql security definer set search_path = public, extensions as $$
declare r record; t text;
begin
  select u.username, u.role into r from users u join user_secrets s using (username)
   where u.username = lower(trim(p_user)) and s.pw_hash = crypt(p_pass, s.pw_hash);
  if not found then return null; end if;
  t := replace(gen_random_uuid()::text, '-', '') || replace(gen_random_uuid()::text, '-', '');
  insert into sessions (token, username) values (t, r.username);
  return json_build_object('username', r.username, 'role', r.role, 'token', t);
end $$;
create or replace function public.app_whoami(p_token text) returns json language sql security definer set search_path = public as $$
  select json_build_object('username', u.username, 'role', u.role) from sessions s join users u using (username) where s.token = p_token $$;
create or replace function public.app_logout(p_token text) returns void language sql security definer set search_path = public as $$
  delete from sessions where token = p_token $$;

create or replace function public.staff_create(p_token text, n_user text, n_pass text, n_role text) returns void language plpgsql security definer set search_path = public, extensions as $$
begin
  if not _admin(p_token) then raise exception 'Not allowed'; end if;
  if n_role not in ('producer','gatekeeper') then raise exception 'Invalid role'; end if;
  if length(trim(n_user)) < 3 or length(n_pass) < 6 then raise exception 'Username needs 3+ characters and password 6+'; end if;
  insert into users (username, role) values (lower(trim(n_user)), n_role);
  insert into user_secrets (username, pw_hash) values (lower(trim(n_user)), crypt(n_pass, gen_salt('bf')));
end $$;
create or replace function public.staff_delete(p_token text, t_user text) returns void language plpgsql security definer set search_path = public as $$
begin
  if not _admin(p_token) then raise exception 'Not allowed'; end if;
  if lower(t_user) = (select username from sessions where token = p_token) then raise exception 'You cannot delete your own account'; end if;
  delete from users where username = lower(t_user);
end $$;
create or replace function public.staff_password(p_token text, t_user text, n_pass text) returns void language plpgsql security definer set search_path = public, extensions as $$
begin
  if not _admin(p_token) then raise exception 'Not allowed'; end if;
  if length(n_pass) < 6 then raise exception 'Password needs 6+ characters'; end if;
  update user_secrets set pw_hash = crypt(n_pass, gen_salt('bf')) where username = lower(t_user);
end $$;

-- continuous numbering: highest existing number per tier, reserved atomically
create or replace function public.reserve_seq(p_token text, p_tier text, p_qty int) returns int language plpgsql security definer set search_path = public as $$
declare cur int;
begin
  if not _staff(p_token) then raise exception 'Not allowed'; end if;
  if p_tier not in ('Ordinary','VIP','VVIP') or p_qty < 1 or p_qty > 5000 then raise exception 'Bad request'; end if;
  perform pg_advisory_xact_lock(hashtext('seq_' || p_tier));
  select coalesce(max(regexp_replace(number, '^.*-', '')::int), 0) into cur from tickets where tier = p_tier;
  cur := greatest(cur, coalesce((select last from ticket_counters where tier = p_tier), 0));
  insert into ticket_counters (tier, last) values (p_tier, cur + p_qty) on conflict (tier) do update set last = cur + p_qty;
  return cur + 1;
end $$;
create or replace function public.purge_tickets(p_token text) returns void language plpgsql security definer set search_path = public as $$
begin
  if not _admin(p_token) then raise exception 'Not allowed'; end if;
  delete from tickets where true; delete from ticket_counters where true;
end $$;

-- digital ticket pages (guests, no login) and shared ticket design
create or replace function public.digital_ticket(p_number text, p_code text) returns json language sql security definer set search_path = public as $$
  select row_to_json(t) from (select number, tier, buyer, code, status from tickets where number = p_number and code = p_code) t $$;
create or replace function public.public_design() returns jsonb language sql security definer set search_path = public as $$
  select value from settings where key = 'design' $$;
create or replace function public.save_design(p_token text, p_design jsonb) returns void language plpgsql security definer set search_path = public as $$
begin
  if not _staff(p_token) then raise exception 'Not allowed'; end if;
  insert into settings (key, value) values ('design', p_design) on conflict (key) do update set value = p_design;
end $$;
